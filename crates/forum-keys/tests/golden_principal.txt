# forum_keys::normalise_principal (CS-1, design D.11.14). Its rules are frozen once released and
# have no version, because a change would split one human into two principals: never change an
# existing row after the release. New rows may be added at any time.
#
# Format: <input> TAB <ok | err> TAB <principal key | KeyError variant>. Lines starting with '#'
# and blank lines are ignored. Both text columns understand the escapes \\ \t \n \r and \u{HEX};
# a space at either end of a column is spelt \u{20}.

# --- The reserved principal ---------------------------------------------------------------------------
user	ok	user
USER	ok	user
\u{20}User\t	ok	user

# --- mail: a bare address, lower-cased whole (a person, unlike a message id) ----------------------
mail:alice@corp.example	ok	mail:alice@corp.example
mail:Alice@Corp.Example	ok	mail:alice@corp.example
MAIL:ALICE@CORP.EXAMPLE	ok	mail:alice@corp.example
mail:caf\u{65}\u{301}@corp.example	ok	mail:caf\u{e9}@corp.example
mail:\u{c9}LODIE@CORP.EXAMPLE	ok	mail:\u{e9}lodie@corp.example
mail:<alice@corp.example>	ok	mail:alice@corp.example
mail:<\u{20}Alice@Corp.Example\u{20}>	ok	mail:alice@corp.example

# --- Any other scheme: lower-cased scheme, the value as it is -------------------------------------------
slack:T01/U02ABC	ok	slack:T01/U02ABC
Slack:T01/U02ABC	ok	slack:T01/U02ABC
team:Payments	ok	team:Payments
github:Octo-Cat	ok	github:Octo-Cat

x:a:b	ok	x:a:b
team:\u{600}x	ok	team:\u{600}x
team:\u{2764}\u{fe0f}	ok	team:\u{2764}\u{fe0f}
# The variation selectors, the Mongolian free variation selectors among them, are the only
# default-ignorable code points kept.
github:octo\u{fe00}cat	ok	github:octo\u{fe00}cat
team:a\u{e0100}	ok	team:a\u{e0100}
github:octo\u{180b}cat	ok	github:octo\u{180b}cat
team:\u{1828}\u{180c}\u{180d}\u{180f}	ok	team:\u{1828}\u{180c}\u{180d}\u{180f}
# One that chooses no glyph for the character before it shows nothing, and is kept all the same:
# team:alice and team:alice\u{180b} read alike and are two principals.
team:alice\u{180b}	ok	team:alice\u{180b}
team:alice\u{fe00}	ok	team:alice\u{fe00}
# Whitespace at the edges, U+0085 and U+2029 among it, is trimmed.
\u{20}slack:T01/U02\u{85}	ok	slack:T01/U02
slack:T01/U02\u{2029}	ok	slack:T01/U02
